To let Triton serve models from this PVC, mount the same claim in your
SuperSONIC release and enable explicit model control:

  triton:
    modelRepository:
      enabled: true
      storageType: pvc
      mountPath: /models
      pvc:
        claimName: {{ .Values.modelRepository.claimName }}
    args:
      - |
        /opt/tritonserver/bin/tritonserver \
        --model-repository=/models \
        --model-control-mode=explicit \
        --load-model=*
